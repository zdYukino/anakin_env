cmake_minimum_required(VERSION 3.16)
project(anakin_env VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(anakin_sdk 0.1.0 EXACT REQUIRED)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/anakin_sdk.revision" SDK_LOCK LIMIT_COUNT 1)
if(NOT SDK_LOCK STREQUAL ANAKIN_SDK_REVISION)
  message(FATAL_ERROR "Installed anakin_sdk revision differs from anakin_sdk.revision")
endif()
find_package(unitree_sdk2 REQUIRED)
add_library(anakin_gate STATIC src/gate.cpp src/native_codec.cpp)
target_include_directories(anakin_gate PUBLIC include)
ament_target_dependencies(anakin_gate anakin_sdk)
target_link_libraries(anakin_gate unitree_sdk2)
find_package(Threads REQUIRED)
add_library(anakin_transport STATIC src/transport.cpp)
target_include_directories(anakin_transport PUBLIC include)
ament_target_dependencies(anakin_transport anakin_sdk)
target_link_libraries(anakin_transport anakin_gate Threads::Threads)
add_executable(anakin_env_native_node src/native_node.cpp)
target_link_libraries(anakin_env_native_node anakin_transport)
ament_target_dependencies(anakin_env_native_node rclcpp anakin_sdk)
add_executable(anakin_env_native_core_test tests/native_core.cpp)
target_link_libraries(anakin_env_native_core_test anakin_transport)
ament_target_dependencies(anakin_env_native_core_test anakin_sdk)
find_package(X11 REQUIRED)
add_executable(anakin_sim_runner tests/sim_runner.cpp)
target_link_libraries(anakin_sim_runner Threads::Threads)
add_executable(anakin_sim_probe tests/sim_probe.cpp)
target_link_libraries(anakin_sim_probe anakin_transport X11::X11 ${CMAKE_DL_LIBS})
ament_target_dependencies(anakin_sim_probe rclcpp anakin_sdk)
add_executable(anakin_env_node src/node.cpp)
target_link_libraries(anakin_env_node anakin_gate)
ament_target_dependencies(anakin_env_node rclcpp anakin_sdk)
add_executable(anakin_env_contract_test tests/contracts.cpp)
target_link_libraries(anakin_env_contract_test anakin_gate)
ament_target_dependencies(anakin_env_contract_test anakin_sdk)
if(BUILD_TESTING)
 enable_testing()
 add_test(NAME execution_contracts COMMAND anakin_env_contract_test)
  add_test(NAME native_control_contracts COMMAND anakin_env_native_core_test)
endif()
add_executable(anakin_env_protocol_fixture tests/protocol_fixture.cpp)
target_include_directories(anakin_env_protocol_fixture PRIVATE include)
ament_target_dependencies(anakin_env_protocol_fixture rclcpp anakin_sdk)
install(TARGETS anakin_sim_runner anakin_sim_probe anakin_env_native_node anakin_env_native_core_test anakin_env_node anakin_env_contract_test anakin_env_protocol_fixture DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY docs/ DESTINATION share/${PROJECT_NAME}/docs)
install(FILES LICENSE LICENSE-Anakin NOTICE.md DESTINATION share/${PROJECT_NAME})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/runtime-sdk.revision" "${SDK_LOCK}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/runtime-sdk.revision" DESTINATION share/${PROJECT_NAME})
ament_package()
